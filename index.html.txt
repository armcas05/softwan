def calculate\_member\_discount(total\_amount, member\_level): # Algorithm 1: Member Discount Calculation if member\_level == 'VIP': discount\_rate = 0.15 elif member\_level == 'Gold': discount\_rate = 0.10 elif member\_level == 'General': discount\_rate = 0.05 else: discount\_rate = 0.0 discount\_amount = total\_amount \* discount\_rate net\_amount = total\_amount - discount\_amount return discount\_amount, net\_amount
CREATE DATABASE IF NOT EXISTS coffeeshop\_db; USE coffeeshop\_db; CREATE TABLE Customer ( customer\_id INT AUTO\_INCREMENT PRIMARY KEY, name VARCHAR(100) NOT NULL, phone VARCHAR(20) UNIQUE NOT NULL, member\_level VARCHAR(20) DEFAULT 'General', accumulated\_points INT DEFAULT 0 ); CREATE TABLE Category ( category\_id INT AUTO\_INCREMENT PRIMARY KEY, category\_name VARCHAR(50) NOT NULL ); CREATE TABLE Product ( product\_id INT AUTO\_INCREMENT PRIMARY KEY, product\_name VARCHAR(100) NOT NULL, price DECIMAL(10,2) NOT NULL, stock\_qty INT NOT NULL, category\_id INT, FOREIGN KEY (category\_id) REFERENCES Category(category\_id) );
